cmake_minimum_required(VERSION 3.20)
project(fintech_ledger_kernel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ledger_kernel src/money.cpp src/journal.cpp)
target_include_directories(ledger_kernel PUBLIC include)
target_compile_options(ledger_kernel PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(kernel_tests tests/kernel_tests.cpp)
target_link_libraries(kernel_tests PRIVATE ledger_kernel)
enable_testing()
add_test(NAME kernel_tests COMMAND kernel_tests)
